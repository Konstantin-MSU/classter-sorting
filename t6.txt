#define A 4

class Tpair {
	// 212-Sushchev-pairs
private:
	float x;
	float y; // координаты
	float d = 100000; // удаленность от центра графа
	int CN = 0; // номер клетки, в которой находится точка
public:
	Tpair(float a, float b) : x(a), y(b) {}

	// геттеры и сеттеры

	float getX() const { return x; }
	float getY() const { return y; }
	float getD() const { return d; }
	int getCN() const { return CN; }

	void setX(float newX) { x = newX; }
	void setY(float newY) { y = newY; }
	void setD(float newD) { d = newD; }
	void setCN(int newc) { CN = newc; }
	Tpair() = default;
};

Tpair ad(Tpair a, Tpair b, float i) {
	// 212-Sushchev - new mean of the set of pairs
	float x = i * a.getX() + b.getX();
	float y = i * a.getY() + b.getY();
	return Tpair(x / (i + 1), y / (i + 1));
}

class Cell {
	// 212-Sushchev - class of squares
private:
	std::vector <Tpair> P;
	int empty = 0; // есть ли точки внутри
	float Ub;
	float Db;
	float Lb;
	float Rb; // границы квадрата
	int cl_num = 0;
public:

	Cell() = default;
	// геттеры и сеттеры

	float getUb() const { return Ub; }
	float getDb() const { return Db; }
	float getLb() const { return Lb; }
	float getRb() const { return Rb; }

	int getE() const { return empty; }
	int getCl() const { return cl_num; }
	std::vector<Tpair> getIn() const { return P; }

	void setUb(float newu) { Ub = newu; }
	void setDb(float newd) { Db = newd; }
	void setLb(float newl) { Lb = newl; }
	void setRb(float newr) { Rb = newr; }
	void setE(int newe) { empty = newe; }
	void setCl(int newc) { cl_num = newc; }


	void addP(Tpair T) {P.push_back(T); }
	
};

void Neighbour(std::vector<Cell>& net, int I, int M, int N) { // пережаем массив, координаты рабочей точки, размер строк
	// 212-Сущев-функция поиска соседей
	net[I].setCl(M);
	// проверяем левый столбик
	if ((I - N > 1) && (net[I - N - 1].getE() != 0)) {
		Neighbour(net, I - N - 1, M, N);
	}
	if ((I - N > 0) && (net[I - N].getE() != 0)) {
		Neighbour(net, I - N, M, N);
	}
	if ((I - N + 1 > 0) && (net[I - N + 1].getE() != 0)) {
		Neighbour(net, I - N + 1, M, N);
	}

	// центральный столбик
	if ((I - 1 > 0) && (net[I - 1].getE() != 0)) {
		Neighbour(net, I - 1, M, N);
	}
	if ((I + 1 < N*N-1) && (net[I + 1].getE() != 0)) {
		Neighbour(net, I + 1, M, N);
	}

	// проверяем правый столбик
	if ((I + N - 1 < N * N - 1) && (net[I + N - 1].getE() != 0)) {
		Neighbour(net, I + N - 1, M, N);
	}
	if ((I + N < N * N - 1) && (net[I + N].getE() != 0)) {
		Neighbour(net, I + N, M, N);
	}
	if ((I + N + 1 < N * N - 1) && (net[I + N + 1].getE() != 0)) {
		Neighbour(net, I + N + 1, M, N);
	}
}

float Sdis(Tpair a, Tpair b) {
	// 212-Sushchev - returns squared distans beetwen 2 pairs

	float x = a.getX() - b.getX();
	float y = a.getY() - b.getY();
	return x * x + y * y;
}
